2. 代码预览测试

点击按钮在右侧预览面板显示代码：

3. 控制台预览测试

点击按钮在右侧预览面板显示日志：

4. 快捷键测试

按 Ctrl+P 切换预览面板

5. 代码示例

// 在控制台中测试（F12 打开开发者工具）
// 1. 显示网页预览
window.echora?.preview?.show({ type: 'url', url: 'https://www.baidu.com', title: '百度' })

// 2. 显示 HTML 预览
window.echora?.preview?.show({ 
  type: 'html', 
  html: '<h1>Hello World</h1><p style="color: blue;">这是 HTML 预览</p>',
  title: 'HTML 测试'
})

// 3. 显示代码预览
window.echora?.preview?.show({
  type: 'file',
  content: 'function hello() {\n  console.log("Hello!");\n}',
  language: 'javascript',
  path: 'test.js'
})

// 4. 显示控制台日志
window.echora?.preview?.show({
  type: 'console',
  logs: [
    { level: 'info', message: '应用启动', timestamp: Date.now() },
    { level: 'warn', message: '内存使用率较高', timestamp: Date.now() },
    { level: 'error', message: '连接失败', timestamp: Date.now() }
  ]
})

// 5. 关闭预览面板
window.echora?.preview?.hide()
